Solution to Challenge 1

# Filter out rows with missing experiment_id
fragment <- day1_data |> 
  filter(!is.na(experiment_id)) |> 
  # Parse timestamp and extract valid fragments from comment
  mutate(
    time = parse_date_time(timestamp, orders = c("ymd HMS", "Ymd HMS")),
    fragment = ifelse(str_detect(comment, "^[A-Za-z0-9]+$"), comment, NA_character_)
  )  |> 
  # Keep only rows with valid time and fragment
  filter(!is.na(time), !is.na(fragment))

Solution to Challenge 3

# Step 1: Filter valid tags
valid_points <- coordinates |> 
  filter(!is.na(tag)) |> 
  filter(str_detect(tag, "^[A-Za-z0-9=]+$"))

# Step 2: Plot valid points
ggplot(valid_points, aes(x = lon, y = lat, label = tag)) +
  geom_point(color = "blue", size = 3) +
  geom_text(vjust = -0.5) +
  theme_minimal() +
  labs(title = "Field Sites — Valid Tags Only",
       subtitle = "Observe the west-to-east alignment to reveal the fragment",
       x = "Longitude", y = "Latitude")

# Step 3: Combine tags from west to east
fragment <- valid_points |> 
  arrange(lon) |>        # order from west (lowest lon) to east (highest lon)
  pull(tag) |>           # extract the tags
  paste(collapse = "")   # combine into one string


# Step 4: Print the resulting fragment
fragment

Solution to Challenge 4

# Step 1: Assign the encoded fragments
encoded1 <- "U09NRVRISU5H"
encoded2 <- "SElEREVO"
encoded3 <- "SU4gS0lMQlVSTg=="

# Step 2: Decode each
decoded1 <- encoded1 |> base64decode() |> rawToChar()
decoded2 <- encoded2 |> base64decode() |> rawToChar()
decoded3 <- encoded3 |> base64decode() |> rawToChar()

# Step 3: Combine into one sentence
final_message <- paste(decoded1, decoded2, decoded3)
